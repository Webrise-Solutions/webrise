# A named user-agent group REPLACES the wildcard group for that crawler — it
# does not add to it. Googlebot and Bingbot therefore need their own Disallow
# lines; without them, listing those crawlers here was quietly granting them
# the /admin access the wildcard group denies everyone else.

User-agent: Googlebot
Allow: /
Disallow: /admin
Disallow: /unsubscribe

User-agent: Bingbot
Allow: /
Disallow: /admin
Disallow: /unsubscribe

Disallow: /admin

Disallow: /admin

Allow: /
Disallow: /admin
